# 🚀 FUNÇÃO PARA ADICIONAR E-MAIL À LISTA NO BREVO
def adicionar_email_lista_brevo(email_usuario, nome_usuario="Novo Cliente"):
    configuration = sib_api_v3_sdk.Configuration()
    configuration.api_key['api-key'] = BREVO_API_KEY

    api_instance = sib_api_v3_sdk.ContactsApi(sib_api_v3_sdk.ApiClient(configuration))
    contato = {"email": email_usuario, "listIds": [int(BREVO_LIST_ID)], "attributes": {"NOME": nome_usuario}}

    try:
        api_instance.create_contact(contato)
        print(f"✅ E-mail {email_usuario} adicionado com sucesso na lista do Brevo.")
    except ApiException as e:
        print(f"❌ Erro ao adicionar e-mail na lista do Brevo: {e}")

# 🚀 FUNÇÃO PARA ENVIAR O E-MAIL COM O MANUAL EM ANEXO
def enviar_email_brevo(destinatario_email, destinatario_nome):
    configuration = sib_api_v3_sdk.Configuration()
    configuration.api_key['api-key'] = BREVO_API_KEY

    api_instance = sib_api_v3_sdk.TransactionalEmailsApi(sib_api_v3_sdk.ApiClient(configuration))
    email = sib_api_v3_sdk.SendSmtpEmail(
        to=[{"email": destinatario_email, "name": destinatario_nome}],
        sender={"email": EMAIL_REMETENTE, "name": NOME_REMETENTE},
        subject=ASSUNTO_EMAIL,
        html_content="""
        <h3>Obrigado por se cadastrar!</h3>
        <p>Olá {nome},</p>
        <p>Estamos felizes em tê-lo conosco! Como prometido, aqui está o seu manual em anexo.</p>
        <p>Caso tenha dúvidas, estamos à disposição.</p>
        """
    )

    if CAMINHO_ANEXO:
        with open(CAMINHO_ANEXO, "rb") as f:
            email.attachment = [sib_api_v3_sdk.SendSmtpEmailAttachment(name="Manual_Planner_Organizer.pdf", content=f.read().encode("base64"))]

    try:
        api_instance.send_transac_email(email)
        print("✅ E-mail enviado com sucesso!")
    except ApiException as e:
        print(f"❌ Erro ao enviar o e-mail: {e}")
