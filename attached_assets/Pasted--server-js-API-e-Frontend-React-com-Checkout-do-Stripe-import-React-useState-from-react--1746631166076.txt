// server.js - API e Frontend React com Checkout do Stripe
import React, { useState } from "react";
import axios from "axios";
import express from "express";
import stripePackage from "stripe";
import path from "path";

const app = express();
app.use(express.json());

// Configuração Stripe
const stripe = stripePackage(process.env.STRIPE_SECRET_KEY);

// Servir o Frontend React
app.use(express.static(path.join(__dirname, "public")));

// Endpoint para criar sessão de checkout do Stripe
app.post("/api/create-checkout-session", async (req, res) => {
    const { planType } = req.body;

    const prices = {
        mensal: "preco_mensal_stripe_id",
        anual: "preco_anual_stripe_id",
        vitalicio: "preco_vitalicio_stripe_id"
    };

    try {
        const session = await stripe.checkout.sessions.create({
            payment_method_types: ["card"],
            line_items: [
                { price: prices[planType], quantity: 1 }
            ],
            mode: planType === "vitalicio" ? "payment" : "subscription",
            success_url: "https://seusite.com/sucesso",
            cancel_url: "https://seusite.com/cancelado"
        });

        res.json({ url: session.url });
    } catch (error) {
        res.status(500).json({ error: "Erro ao criar sessão de checkout" });
    }
});

// Rota de Frontend (React)
app.get("/*", (req, res) => {
    res.sendFile(path.join(__dirname, "public", "index.html"));
});

// Inicializando o servidor
const PORT = process.env.PORT || 3000;
app.listen(PORT, () => console.log(`Servidor rodando na porta ${PORT}`));

// Frontend React
const CheckoutButtons = () => {
    const [loading, setLoading] = useState(false);

    const handleCheckout = async (planType) => {
        setLoading(true);
        try {
            const response = await axios.post("/api/create-checkout-session", { planType });
            if (response.data.url) {
                window.location.href = response.data.url;
            } else {
                alert("Erro ao iniciar o checkout. Tente novamente.");
            }
        } catch (error) {
            console.error("Erro ao redirecionar para o Stripe:", error);
            alert("Erro ao iniciar o checkout. Tente novamente.");
        } finally {
            setLoading(false);
        }
    };

    return (
        <div className="flex flex-col space-y-4">
            <button onClick={() => handleCheckout("mensal")} disabled={loading} className="w-full text-white bg-blue-500 hover:bg-blue-600 rounded-xl py-2">
                {loading ? "Processando..." : "Assinar Mensal"}
            </button>

            <button onClick={() => handleCheckout("anual")} disabled={loading} className="w-full text-white bg-red-500 hover:bg-red-600 rounded-xl py-2">
                {loading ? "Processando..." : "Assinar Anual"}
            </button>

            <button onClick={() => handleCheckout("vitalicio")} disabled={loading} className="w-full text-white bg-green-500 hover:bg-green-600 rounded-xl py-2">
                {loading ? "Processando..." : "Comprar Vitalício"}
            </button>
        </div>
    );
};

export default CheckoutButtons;
