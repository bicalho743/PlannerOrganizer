from reportlab.lib.pagesizes import A4
from reportlab.pdfgen import canvas
from reportlab.lib import colors

def gerar_pdf_proposta_59():
    arquivo_pdf = "Proposta_59_Lilian_Mara_de_Bernardi_Costa.pdf"
    c = canvas.Canvas(arquivo_pdf, pagesize=A4)
    largura, altura = A4

    # Cores personalizadas
    cinza_claro = colors.HexColor("#808080")     # fundo
    cinza_medio = colors.HexColor("#696969")     # textos
    verde_escuro = colors.HexColor("#008000")    # acentos
    verde_claro = colors.HexColor("#90EE90")     # destaque

    # Fundo
    c.setFillColor(cinza_claro)
    c.rect(0, 0, largura, altura, fill=True, stroke=0)

    # Cabeçalho
    c.setFillColor(verde_escuro)
    c.rect(0, altura - 60, largura, 60, fill=True, stroke=0)
    c.setFillColor(colors.white)
    c.setFont("Helvetica-Bold", 18)
    c.drawString(30, altura - 40, "Proposta #59 – Lilian Mara de Bernardi Costa")

    # Data
    c.setFillColor(cinza_medio)
    c.setFont("Helvetica", 10)
    c.drawString(30, altura - 70, "Data: 24/04/2025")

    # Bloco de informações
    y = altura - 100
    c.setFont("Helvetica-Bold", 12)
    c.drawString(30, y, "Informações da Proposta")
    y -= 20
    c.setFont("Helvetica", 11)
    c.drawString(40, y, "Tipo: Organização")
    y -= 16
    c.drawString(40, y, "Status: Em elaboração")
    y -= 16
    c.drawString(40, y, "Data Início: 21/04/2025")
    y -= 16
    c.drawString(40, y, "Data Fim: 26/04/2025")
    y -= 16
    c.drawString(40, y, "Prazo de Entrega: 5 dias")

    # Bloco de investimento
    y -= 30
    c.setFillColor(verde_claro)
    c.rect(30, y - 40, largura - 60, 60, fill=True, stroke=0)
    c.setFillColor(cinza_medio)
    c.setFont("Helvetica-Bold", 12)
    c.drawString(40, y + 10, "Investimento")
    c.setFont("Helvetica", 12)
    c.drawString(40, y - 10, "• Mudança")
    c.drawString(150, y - 10, "• Pós-mudança")
    c.drawString(280, y - 10, "• Consultoria")
    c.setFont("Helvetica-Bold", 12)
    c.setFillColor(verde_escuro)
    c.drawString(40, y - 30, "Total: R$ 4.000,00 – Status: Pendente")

    # Observações
    y -= 80
    c.setFillColor(cinza_medio)
    c.setFont("Helvetica-Bold", 12)
    c.drawString(30, y, "Observações:")
    c.setFont("Helvetica", 11)
    c.setFillColor(colors.black)
    y -= 20
    c.drawString(40, y, "• Os valores apresentados incluem todos os custos e acréscimos.")
    y -= 16
    c.drawString(40, y, "• Valores a receber incluem base e serviços de organização.")
    y -= 16
    c.drawString(40, y, "• Valores a pagar a lojas/fornecedores são responsabilidade do cliente.")

    # Rodapé
    c.setFillColor(verde_escuro)
    c.rect(0, 0, largura, 40, fill=True, stroke=0)
    c.setFillColor(colors.white)
    c.setFont("Helvetica", 10)
    c.drawString(30, 15, "Planner Organizer | contato@plannerorganizer.com.br | www.plannerorganizer.com.br")

    c.save()
    print(f"PDF gerado: {arquivo_pdf}")
