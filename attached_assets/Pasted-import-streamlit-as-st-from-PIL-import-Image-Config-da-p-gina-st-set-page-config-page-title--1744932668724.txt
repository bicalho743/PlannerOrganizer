import streamlit as st
from PIL import Image

# Config da página
st.set_page_config(page_title="PlannerOrganizer", page_icon="📋", layout="centered")

# Inicializa sessão
if "logado" not in st.session_state:
    st.session_state.logado = False

# Estilo customizado
st.markdown("""
    <style>
        .stButton > button {
            background-color: #2d8cff;
            color: white;
            border-radius: 12px;
            padding: 0.6rem 1.2rem;
            font-weight: bold;
            font-size: 16px;
        }
        .stButton > button:hover {
            background-color: #105cb6;
        }
    </style>
""", unsafe_allow_html=True)

# Se já estiver logado, vai direto para o painel
if st.session_state.logado:
    st.switch_page("painel.py")  # <-- funciona com Streamlit 1.20+ e múltiplas páginas

# Página inicial
st.image("https://i.imgur.com/0uZrDxy.png", width=200)
st.markdown("<h1 style='text-align: center;'>Organize sua carreira com inteligência</h1>", unsafe_allow_html=True)
st.markdown("<h4 style='text-align: center; color: #666;'>O app ideal para quem é Personal Organizer e quer crescer com estratégia</h4>", unsafe_allow_html=True)
st.markdown("---")

st.markdown("### ✨ Por que usar o PlannerOrganizer?")
st.markdown("- 📊 Controle financeiro por cliente")
st.markdown("- 🧾 Propostas profissionais em segundos")
st.markdown("- 💰 Precificação inteligente")
st.markdown("- 📈 Relatórios automáticos do seu negócio")

st.markdown("---")
st.markdown("### Faça login para começar:")

col1, col2, col3 = st.columns(3)
with col1:
    if st.button("🔐 Login com E-mail"):
        st.session_state.logado = True
        st.switch_page("painel.py")
with col2:
    if st.button("🔵 Google"):
        st.session_state.logado = True
        st.switch_page("painel.py")
with col3:
    if st.button("🔵 Facebook"):
        st.session_state.logado = True
        st.switch_page("painel.py")

st.markdown("---")
