import stripe
from fastapi import FastAPI
from fastapi.responses import RedirectResponse, JSONResponse
from fastapi.middleware.cors import CORSMiddleware
from starlette.staticfiles import StaticFiles

# Sua chave secreta do Stripe
stripe.api_key = "sk_test_SUA_CHAVE_SECRETA"

app = FastAPI()

# Middleware CORS para permitir chamadas do frontend
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)

# Servir arquivos HTML da pasta /static
app.mount("/static", StaticFiles(directory="static"), name="static")

@app.get("/")
def home():
    return RedirectResponse(url="/static/index.html")

# Plano MENSAL - Sem trial
@app.post("/checkout/mensal")
def checkout_mensal():
    session = stripe.checkout.Session.create(
        payment_method_types=["card"],
        mode="subscription",
        line_items=[{
            "price": "price_1RFBNXLWUPER7pUXzmz8cdsL",  # R$ 9,70
            "quantity": 1,
        }],
        success_url="https://plannerorganizer.com.br/sucesso",
        cancel_url="https://plannerorganizer.com.br/erro",
    )
    return JSONResponse({"id": session.id})
