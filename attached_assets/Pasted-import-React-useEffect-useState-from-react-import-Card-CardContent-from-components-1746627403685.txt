import React, { useEffect, useState } from "react";
import { Card, CardContent } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import axios from "axios";

const LandingPage = () => {
    const [loading, setLoading] = useState(false);

    const handleCheckout = async (planType) => {
        setLoading(true);
        try {
            const response = await axios.post("/api/create-checkout-session", { planType });
            window.location.href = response.data.url;
        } catch (error) {
            console.error("Erro ao redirecionar para o Stripe:", error);
            alert("Erro ao iniciar o checkout. Tente novamente.");
        } finally {
            setLoading(false);
        }
    };

    return (
        <div className="flex flex-col items-center justify-center min-h-screen bg-gray-100">
            <div className="flex gap-4">
                <Card className="w-64 p-4 text-center shadow-lg">
                    <CardContent>
                        <h2 className="text-xl font-bold">Plano Mensal</h2>
                        <p className="text-3xl font-bold mb-4">R$9,70</p>
                        <Button onClick={() => handleCheckout('mensal')} disabled={loading} className="w-full text-white bg-blue-500 hover:bg-blue-600 rounded-xl py-2">{loading ? "Processando..." : "Assinar Mensal"}</Button>
                    </CardContent>
                </Card>

                <Card className="w-64 p-4 text-center shadow-lg">
                    <CardContent>
                        <h2 className="text-xl font-bold">Plano Anual</h2>
                        <p className="text-3xl font-bold mb-4">R$97,00</p>
                        <Button onClick={() => handleCheckout('anual')} disabled={loading} className="w-full text-white bg-red-500 hover:bg-red-600 rounded-xl py-2">{loading ? "Processando..." : "Assinar Anual"}</Button>
                    </CardContent>
                </Card>

                <Card className="w-64 p-4 text-center shadow-lg">
                    <CardContent>
                        <h2 className="text-xl font-bold">Acesso Vitalício</h2>
                        <p className="text-3xl font-bold mb-4">R$247,00</p>
                        <Button onClick={() => handleCheckout('vitalicio')} disabled={loading} className="w-full text-white bg-blue-500 hover:bg-blue-600 rounded-xl py-2">{loading ? "Processando..." : "Comprar Vitalício"}</Button>
                    </CardContent>
                </Card>
            </div>
        </div>
    );
};

export default LandingPage;
