import sib_api_v3_sdk
from sib_api_v3_sdk.rest import ApiException
import os

# Configurar a API Key do Brevo (Sendinblue)
api_key = os.getenv("BREVO_API_KEY")  # Defina a chave de API como variável de ambiente
lista_brevo_id = os.getenv("BREVO_LIST_ID")  # ID da lista do Brevo onde os e-mails serão armazenados

def adicionar_email_lista_brevo(email_usuario, nome_usuario="Novo Cliente"):
    configuration = sib_api_v3_sdk.Configuration()
    configuration.api_key['api-key'] = api_key

    api_instance = sib_api_v3_sdk.ContactsApi(sib_api_v3_sdk.ApiClient(configuration))
    
    # Dados do contato a ser adicionado
    contato = {
        "email": email_usuario,
        "listIds": [int(lista_brevo_id)],  # Lista específica no Brevo
        "attributes": {"NOME": nome_usuario}
    }
    
    try:
        api_instance.create_contact(contato)
        print(f"✅ E-mail {email_usuario} adicionado com sucesso na lista do Brevo.")
    except ApiException as e:
        print(f"❌ Erro ao adicionar e-mail na lista do Brevo: {e}")

def enviar_email_brevo(destinatario_email, destinatario_nome, assunto, mensagem_html, anexo=None):
    configuration = sib_api_v3_sdk.Configuration()
    configuration.api_key['api-key'] = api_key

    api_instance = sib_api_v3_sdk.TransactionalEmailsApi(sib_api_v3_sdk.ApiClient(configuration))
    email = sib_api_v3_sdk.SendSmtpEmail(
        to=[{"email": destinatario_email, "name": destinatario_nome}],
        sender={"email": "seuemail@seusite.com.br", "name": "Sua Empresa"},
        subject=assunto,
        html_content=mensagem_html
    )
    
    if anexo:
        with open(anexo, "rb") as f:
            email.attachment = [sib_api_v3_sdk.SendSmtpEmailAttachment(
                name="Manual.pdf",
                content=f.read().encode("base64")
            )]
    
    try:
        api_response = api_instance.send_transac_email(email)
        print("✅ E-mail enviado com sucesso!")
    except ApiException as e:
        print("❌ Erro ao enviar o e-mail: %s\n" % e)
