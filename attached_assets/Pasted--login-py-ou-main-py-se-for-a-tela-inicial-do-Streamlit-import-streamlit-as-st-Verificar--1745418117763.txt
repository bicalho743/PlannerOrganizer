# login.py (ou main.py se for a tela inicial do Streamlit)

import streamlit as st

# Verificar se o usuário está tentando registrar ou recuperar senha
if st.session_state.get("login_page") == "registrar":
    try:
        from pages.registrar import show
        show()
        st.stop()
    except ImportError as e:
        st.error(f"Não foi possível carregar o módulo de registro: {e}")
        st.session_state.login_page = "login"
        st.rerun()

elif st.session_state.get("login_page") == "recuperar_senha":
    try:
        from pages.recuperar_senha import show
        show()
        st.stop()
    except ImportError as e:
        st.error(f"Não foi possível carregar o módulo de recuperação de senha: {e}")
        st.session_state.login_page = "login"
        st.rerun()

# Código da interface de login normal (email e senha)
with st.form("login_form"):
    email = st.text_input("Email")
    password = st.text_input("Senha", type="password")
    submit = st.form_submit_button("Entrar")

    if submit:
        if email.lower() == "admin" and password == "admin":
            st.session_state.authenticated = True
            st.success("Login realizado com sucesso (modo demonstração)!")
            st.rerun()
        else:
            st.error("Usuário ou senha incorretos")

col1, col2 = st.columns(2)
with col1:
    if st.button("Esqueceu sua senha?"):
        st.session_state.login_page = "recuperar_senha"
        st.rerun()

with col2:
    if st.button("Criar uma conta"):
        st.session_state.login_page = "registrar"
        st.rerun()

# pages/registrar.py

def show():
    import streamlit as st

    st.title("Criar Conta")
    st.markdown("Preencha os campos abaixo para se registrar.")

    with st.form("registrar_form"):
        email = st.text_input("Email")
        password = st.text_input("Senha", type="password")
        confirm = st.text_input("Confirmar senha", type="password")
        submit = st.form_submit_button("Registrar")

        if submit:
            if password != confirm:
                st.error("As senhas não coincidem.")
            elif not email or not password:
                st.error("Preencha todos os campos.")
            else:
                # Integração com Firebase ou backend aqui
                st.success("Conta criada com sucesso! Faça login agora.")
                st.session_state.login_page = "login"
                st.rerun()

    if st.button("Voltar para login"):
        st.session_state.login_page = "login"
        st.rerun()
