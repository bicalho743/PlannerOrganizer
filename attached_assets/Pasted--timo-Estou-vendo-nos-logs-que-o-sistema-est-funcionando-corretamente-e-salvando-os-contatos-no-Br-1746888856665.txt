Código de Integração com Brevo (Node.js):

// Usando a biblioteca Brevo/Sendinblue diretamente
import SibApiV3Sdk from 'sib-api-v3-sdk';
// Chave API do Brevo
const API_KEY = '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';
// ID da lista Brevo - Leads Planner Organizer (ID #7)
const BREVO_LIST_ID = 7;
// Configurar a API do Brevo
const defaultClient = SibApiV3Sdk.ApiClient.instance;
const apiKey = defaultClient.authentications['api-key'];
apiKey.apiKey = API_KEY;
/**
 * Adiciona um contato à lista do Brevo
 * @param email Email do lead
 * @param firstName Nome do lead
 * @param lastName Sobrenome do lead (opcional)
 * @returns Promise com o resultado da operação
 */
export async function addContactToBrevo(
  email: string,
  firstName: string,
  lastName?: string
) {
  try {
    // Criar uma instância da API de contatos
    const apiInstance = new SibApiV3Sdk.ContactsApi();
    
    // Criar um objeto CreateContact do Brevo
    const createContact = new SibApiV3Sdk.CreateContact();
    
    // Configurar os dados do contato
    createContact.email = email;
    createContact.listIds = [BREVO_LIST_ID];
    // Adicionando updateEnabled como opção via any type
    (createContact as any).updateEnabled = true;
    
    // Adicionar atributos do contato
    const attributes = {
      FIRSTNAME: firstName,
      NOME: firstName
    };
    
    // Adicionar sobrenome se fornecido
    if (lastName && lastName.trim() !== '') {
      attributes.LASTNAME = lastName;
      attributes.SOBRENOME = lastName;
    }
    
    // Adicionar fonte para rastreamento
    attributes.SOURCE = 'Landing Page Personal Manager';
    attributes.ORIGEM = 'Landing Page Personal Manager';
    
    // Definir os atributos no objeto de contato
    createContact.attributes = attributes;
    
    // Enviar para o Brevo usando o formato correto esperado pela API
    const result = await apiInstance.createContact(createContact);
    
    return { success: true, data: result };
  } catch (error) {
    console.error('Erro ao adicionar contato ao Brevo:', error);
    
    // Verificar se o contato já existe
    if (error.response && error.response.statusCode === 400 && 
        error.response.text && error.response.text.includes('Contact already exist')) {
      // Contato já existe
      return { success: true, message: 'Contact already exists in Brevo' };
    }
    
    return { 
      success: false, 
      error: error.message || 'Erro desconhecido ao adicionar contato ao Brevo' 
    };
  }
}
